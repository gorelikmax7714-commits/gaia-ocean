# GAYA MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the MVP of GAYA — an offline-first Hebrew-RTL PWA where Gaia (entering 4th grade) practices multiplication, vertical addition/subtraction, English phonics and first words, while raising a sea-unicorn pet; progress syncs to Max's Google Sheet.

**Architecture:** Static PWA (vanilla ES modules, no build step) hosted on GitHub Pages. Pure-logic "engine" modules (exercise generators, progression, SRS, pet) are dependency-injected and unit-tested with `node --test`. Thin DOM "screens" layer renders into `#app`. Progress lives in localStorage; a sync queue POSTs session events to a Google Apps Script Web App that writes to Google Sheets.

**Tech Stack:** HTML/CSS/JS (ES modules), Node ≥ 20 built-in test runner (tests only), Service Worker + Web App Manifest, Web Speech API (speechSynthesis), Google Apps Script + Google Sheets.

**Spec:** `docs/superpowers/specs/2026-07-25-gaia-ocean-design.md`

## Global Constraints

- App name is **GAYA** (Latin letters in logo/titles); all UI copy is Hebrew, `dir="rtl"`; learning content only is English.
- No npm dependencies, no bundler, no build step. `package.json` contains only `{"type": "module"}` so `node --test` runs ES modules.
- Engine modules (`js/engine/**`) are pure: no DOM, no `localStorage`, no `fetch`, no `Date.now()`/`Math.random()` calls — time and RNG are injected parameters.
- Immutability: state-updating functions return new objects, never mutate inputs (project rule, CRITICAL).
- Files ≤ 400 lines; functions ≤ 50 lines.
- No punishment UX: wrong answers never deduct stars; feedback text is gentle ("כמעט! נסי שוב 💪").
- Secrets: the GAS sync token lives in Script Properties on the server side; the client token is a family-grade shared string in `js/config.js` (documented as non-secret).
- Dates use local device time formatted `YYYY-MM-DD` via one shared helper (`localDateISO`).
- Run all tests with: `node --test` from repo root. All tests must pass before every commit.

```js
import { loadState, saveState } from './storage.js';

const screens = new Map();

export function registerScreen(name, renderFn) {
  screens.set(name, renderFn);
}

const KNOWN_SCREENS = ['onboarding', 'lagoon', 'map', 'reward'];

export function resolveRoute(hash, hasProfile) {
  if (!hasProfile) return { screen: 'onboarding', params: {} };
  const parts = (hash || '').replace(/^#/, '').split('/').filter(Boolean);
  if (parts[0] === 'game' && parts.length === 3) {
    return { screen: 'game', params: { region: parts[1], level: Number(parts[2]) } };
  }
  if (KNOWN_SCREENS.includes(parts[0])) return { screen: parts[0], params: {} };
  return { screen: 'lagoon', params: {} };
}

export function navigate(hash) {
  window.location.hash = hash;
}

function renderCurrent(ctx) {
  const route = resolveRoute(window.location.hash, Boolean(ctx.state.profile));
  const render = screens.get(route.screen);
  const root = document.getElementById('app');
  root.innerHTML = '';
  render(root, ctx, route.params);
}

export function boot() {
  const ctx = {
    state: loadState(window.localStorage),
    update(next) {
      ctx.state = next;
      saveState(window.localStorage, next);
    },
  };
  window.addEventListener('hashchange', () => renderCurrent(ctx));
  renderCurrent(ctx);
}
```

```js
import assert from 'node:assert/strict';
import {
  mulExercise, mulMissingExercise, mulPair, verticalExercise, shuffle,
} from '../js/engine/exercises/math.js';

// Deterministic LCG so tests are reproducible without Math.random.
const lcg = (seed) => () => {
  seed = (seed * 48271) % 2147483647;
  return seed / 2147483647;
};

test('mulExercise: factors from tables, correct answer among 4 unique choices', () => {
  const rng = lcg(42);
  for (let i = 0; i < 200; i++) {
    const ex = mulExercise({ tables: [3, 7] }, rng);
    const [a, b] = ex.question.split(' × ').map(Number);
    assert.ok([3, 7].includes(a));
    assert.ok(b >= 1 && b <= 10);
    assert.equal(ex.answer, a * b);
    assert.equal(ex.choices.length, 4);
    assert.equal(new Set(ex.choices).size, 4);
    assert.ok(ex.choices.includes(ex.answer));
    assert.ok(ex.choices.every((c) => c >= 0));
  }
});

test('mulMissingExercise: hidden factor is the answer', () => {
  const rng = lcg(7);
  for (let i = 0; i < 200; i++) {
    const ex = mulMissingExercise({ tables: [4] }, rng);
    const m = ex.question.match(/^\? × (\d+) = (\d+)$/);
    assert.ok(m, `bad question: ${ex.question}`);
    assert.equal(ex.answer * Number(m[1]), Number(m[2]));
    assert.ok(ex.choices.includes(ex.answer));
  }
});

test('verticalExercise: subtraction never goes negative, answers correct', () => {
  const rng = lcg(99);
  for (let i = 0; i < 200; i++) {
    const ex = verticalExercise({ digits: 3, op: '-' }, rng);
    assert.ok(ex.a >= ex.b);
    assert.equal(ex.answer, ex.a - ex.b);
    assert.ok(ex.a <= 999);
  }
  const add = verticalExercise({ digits: 4, op: '+' }, rng);
  assert.equal(add.answer, add.a + add.b);
});

test('shuffle returns new array with same members', () => {

  const src = [1, 2, 3, 4, 5];
  const out = shuffle(src, rng);
  assert.notEqual(out, src);
  assert.deepEqual([...out].sort(), [1, 2, 3, 4, 5]);
  assert.deepEqual(src, [1, 2, 3, 4, 5]); // immutability
});

test('mulPair builds the exact requested exercise', () => {
```

`js/engine/exercises/math.js`:
```js
// Pure exercise generators. rng: () => number in [0,1), injected by caller.

const pick = (arr, rng) => arr[Math.floor(rng() * arr.length)];
const int = (min, max, rng) => min + Math.floor(rng() * (max - min + 1));

export function shuffle(arr, rng) {
  const out = [...arr];
  for (let i = out.length - 1; i > 0; i--) {
    const j = Math.floor(rng() * (i + 1));
    [out[i], out[j]] = [out[j], out[i]];
  }
  return out;
}

function withChoices(ex, rng) {
  const choices = new Set([ex.answer]);
  let guard = 0;
  while (choices.size < 4 && guard < 100) {
    guard += 1;
    const delta = int(1, 10, rng) * (rng() < 0.5 ? -1 : 1);
    const candidate = ex.answer + delta;
    if (candidate >= 0) choices.add(candidate);
  }
  return { ...ex, choices: shuffle([...choices], rng) };
}

export function mulExercise({ tables }, rng) {
  const a = pick(tables, rng);
  const b = int(1, 10, rng);
  return withChoices({ kind: 'bubbles', question: `${a} × ${b}`, answer: a * b }, rng);
}

export function mulMissingExercise({ tables }, rng) {
  const a = pick(tables, rng);
  const b = int(1, 10, rng);
  return withChoices({ kind: 'bubbles', question: `? × ${a} = ${a * b}`, answer: b }, rng);
}

export function mulPair(a, b, rng) {
  return withChoices({ kind: 'bubbles', question: `${a} × ${b}`, answer: a * b }, rng);
}

export function verticalExercise({ digits, op }, rng) {
  const max = 10 ** digits - 1;
  let a = int(10, max, rng);
  let b = int(10, max, rng);
  if (op === '-' && b > a) [a, b] = [b, a];
  return { kind: 'vertical', a, b, op, answer: op === '+' ? a + b : a - b };
}
```

```js
function checkItem(item, path, errors) {
  for (const key of ITEM_KEYS) {
    if (typeof item[key] !== 'string' || item[key].length === 0) {

    }

export function validateContent(json) {
  const errors = [];
  if (!Array.isArray(json.phonics)) errors.push('phonics: must be an array');
  else json.phonics.forEach((p, i) => {
    if (typeof p.letter !== 'string') errors.push(`phonics[${i}]: missing "letter"`);
    if (typeof p.sound !== 'string') errors.push(`phonics[${i}]: missing "sound"`);
    checkItem(p, `phonics[${i}]`, errors);

      errors.push(`${section}: must be an object`);
```

```js
export const REGIONS = [
  { id: 'sounds', he: 'מפרץ הצלילים', emoji: '🐚', unlockStars: 0, levels: 8 },
  { id: 'times', he: 'שונית הכפל', emoji: '🪸', unlockStars: 0, levels: 10 },
  { id: 'words', he: 'אי המילים', emoji: '🏝️', unlockStars: 6, levels: 8 },
  { id: 'vertical', he: 'מערת החיבור', emoji: '🐙', unlockStars: 10, levels: 8 },
];

export function starsForResult({ correct, total }) {
  if (correct === total) return 3;
  if (correct / total >= 0.7) return 2;
  return 1;
}

export function totalStars(progress) {
  return Object.values(progress).reduce(
    (sum, region) => sum + Object.values(region).reduce((s, l) => s + l.stars, 0),
    0,
  );
}

export function isRegionUnlocked(progress, regionId) {
  const region = REGIONS.find((r) => r.id === regionId);
  return Boolean(region) && totalStars(progress) >= region.unlockStars;
}

export function isLevelUnlocked(progress, regionId, levelIndex) {
  if (!isRegionUnlocked(progress, regionId)) return false;
  if (levelIndex === 0) return true;
  return (progress[regionId]?.[levelIndex - 1]?.stars ?? 0) >= 1;
}

export function recordLevelResult(progress, regionId, levelIndex, stars) {
  const prev = progress[regionId]?.[levelIndex] ?? { stars: 0, attempts: 0 };
  return {
    ...progress,
    [regionId]: {
      ...(progress[regionId] ?? {}),
      [levelIndex]: { stars: Math.max(prev.stars, stars), attempts: prev.attempts + 1 },
    },
  };
}

const dayAfter = (iso) => {
  const [y, m, d] = iso.split('-').map(Number);
  const next = new Date(Date.UTC(y, m - 1, d + 1));
  return next.toISOString().slice(0, 10);
};

export function updateStreak(streak, todayISO) {
  if (streak.lastPracticeDate === todayISO) return streak;
  const current = streak.lastPracticeDate && dayAfter(streak.lastPracticeDate) === todayISO
    ? streak.current + 1
    : 1;
  return { current, best: Math.max(streak.best, current), lastPracticeDate: todayISO };
}
```

```js
const dayAfter = (iso) => {
  const [y, m, d] = iso.split('-').map(Number);
  return new Date(Date.UTC(y, m - 1, d + 1)).toISOString().slice(0, 10);
};

export function recordMistake(mistakes, key, todayISO) {
  const rest = mistakes.filter((m) => m.key !== key);
  return [...rest, { key, correctStreak: 0, dueDate: todayISO }];
}

export function recordSuccess(mistakes, key, todayISO) {
  const entry = mistakes.find((m) => m.key === key);
  if (!entry) return mistakes;
  const rest = mistakes.filter((m) => m.key !== key);
  if (entry.correctStreak + 1 >= 2) return rest;
  return [...rest, { ...entry, correctStreak: entry.correctStreak + 1, dueDate: dayAfter(todayISO) }];
}

export function dueMistakes(mistakes, todayISO) {
  return mistakes.filter((m) => m.dueDate <= todayISO).map((m) => m.key);
}
```

```js
const daysBetween = (fromISO, toISO) => {
  const toUTC = (iso) => {
    const [y, m, d] = iso.split('-').map(Number);
    return Date.UTC(y, m - 1, d);
  };
  return Math.round((toUTC(toISO) - toUTC(fromISO)) / 86400000);
```

```js
const KEY = 'gaya-state';

export function defaultState() {
  return {
    profile: null,
    pet: { lastFed: null },
    wallet: { stars: 0 },
    streak: { current: 0, best: 0, lastPracticeDate: null },
    progress: {},
    mistakes: [],
    syncQueue: [],
  };
}

export function loadState(storage) {
  try {
    const raw = storage.getItem(KEY);
    if (!raw) return defaultState();
    const saved = JSON.parse(raw);
    const base = defaultState();
    return {
      ...base,
      ...saved,
      pet: { ...base.pet, ...saved.pet },
      wallet: { ...base.wallet, ...saved.wallet },
      streak: { ...base.streak, ...saved.streak },
    };
  } catch {
    return defaultState();
  }
}

export function saveState(storage, state) {
  try {
    storage.setItem(KEY, JSON.stringify(state));
    return true;
  } catch {
    return false;
  }
}

export function localDateISO(date) {
  const pad = (n) => String(n).padStart(2, '0');
  return `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}`;
}
```

```js
import {
  mulExercise, mulMissingExercise, verticalExercise, shuffle,
} from './engine/exercises/math.js';
import { phonicsItems, cvcWords, vocab } from './engine/content.js';
import { EXERCISES_PER_LEVEL } from './config.js';

const TIMES_LEVELS = [
  [2], [5], [10], [2, 5, 10], [3], [4], [3, 4], [6, 7], [8, 9],
  [2, 3, 4, 5, 6, 7, 8, 9],
];
const VERTICAL_LEVELS = [
  { digits: 2, op: '+' }, { digits: 2, op: '-' },
  { digits: 3, op: '+' }, { digits: 3, op: '-' }, { digits: 3, op: 'mixed' },
  { digits: 4, op: '+' }, { digits: 4, op: '-' }, { digits: 4, op: 'mixed' },
];
const PHONICS_ORDER = 'SATPINMDGOCKEURHBFLJVWXYZQ'.split('');
const WORDS_LEVELS = ['a', 'o', 'u', 'e', 'i', 'mixed', 'sea', 'colors'];

const pick = (arr, rng) => arr[Math.floor(rng() * arr.length)];

function timesLevel(levelIndex, rng) {
  const tables = TIMES_LEVELS[levelIndex];
  const useMissing = levelIndex >= 8;
  return Array.from({ length: EXERCISES_PER_LEVEL }, (_, i) => {
    const gen = useMissing && i % 3 === 2 ? mulMissingExercise : mulExercise;
    const ex = gen({ tables }, rng);
    const [a, b] = ex.question.includes('?')
      ? [Number(ex.question.match(/× (\d+)/)[1]), ex.answer]
      : ex.question.split(' × ').map(Number);
    return { ...ex, srsKey: `mul:${a}x${b}` };
  });
}

function verticalLevel(levelIndex, rng) {
  const cfg = VERTICAL_LEVELS[levelIndex];
  return Array.from({ length: EXERCISES_PER_LEVEL }, () => {
    const op = cfg.op === 'mixed' ? (rng() < 0.5 ? '+' : '-') : cfg.op;
    const ex = verticalExercise({ digits: cfg.digits, op }, rng);
    return { ...ex, srsKey: `vert:${ex.a}${ex.op}${ex.b}` };
  });
}

function listenExercise(item, pool, rng) {
  const wrong = shuffle(pool.filter((p) => p.letter !== item.letter), rng).slice(0, 3);
  return {
    kind: 'listen',
    say: item.sound,
    prompt: 'איזו אות עושה את הצליל?',
    options: shuffle([item.letter, ...wrong.map((w) => w.letter)], rng),
    answer: item.letter,
    srsKey: `phonics:${item.letter}`,
  };
}

function matchExercise(item, pool, rng, srsPrefix) {
  const wrong = shuffle(pool.filter((p) => p.word !== item.word), rng).slice(0, 3);
  return {
    kind: 'match',
    word: item.word,
    emoji: item.emoji,
    he: item.he,
    options: shuffle([item, ...wrong], rng).map(({ word, emoji }) => ({ word, emoji })),
    srsKey: `${srsPrefix}:${item.word}`,
  };
}

function wordbuildExercise(item, rng) {
  const distractor = pick('bcdfghjklmnprstw'.split('').filter((c) => !item.word.includes(c)), rng);
  return {
    kind: 'wordbuild',
    word: item.word,
    he: item.he,
    emoji: item.emoji,
    letters: shuffle([...item.word, distractor], rng),
    srsKey: `cvc:${item.word}`,
  };
}

function soundsLevel(levelIndex, content, rng) {
  const perLevel = Math.ceil(PHONICS_ORDER.length / 8);
  const letters = PHONICS_ORDER.slice(levelIndex * perLevel, (levelIndex + 1) * perLevel);
  const all = phonicsItems(content);
  const items = all.filter((p) => letters.includes(p.letter));
  return Array.from({ length: EXERCISES_PER_LEVEL }, (_, i) => {
    const item = items[i % items.length];
    return i % 2 === 0
      ? listenExercise(item, all, rng)
      : { ...matchExercise(item, all, rng, 'phonics'), srsKey: `phonics:${item.letter}` };
  });
}

function wordsLevel(levelIndex, content, rng) {
  const group = WORDS_LEVELS[levelIndex];
  const vowels = ['a', 'o', 'u', 'e', 'i'];
  const items = group === 'mixed'
    ? vowels.flatMap((v) => cvcWords(content, v))
    : vowels.includes(group) ? cvcWords(content, group) : vocab(content, group);
  const srsPrefix = vowels.includes(group) || group === 'mixed' ? 'cvc' : 'vocab';
  return Array.from({ length: EXERCISES_PER_LEVEL }, (_, i) => {
    const item = items[i % items.length];
    if (srsPrefix === 'vocab' || i % 2 === 1) return matchExercise(item, items, rng, srsPrefix);
    return wordbuildExercise(item, rng);
  });
}

export function buildLevel(regionId, levelIndex, content, rng) {
  if (regionId === 'times') return timesLevel(levelIndex, rng);
  if (regionId === 'vertical') return verticalLevel(levelIndex, rng);
  if (regionId === 'sounds') return soundsLevel(levelIndex, content, rng);
  if (regionId === 'words') return wordsLevel(levelIndex, content, rng);
  throw new Error(`unknown region: ${regionId}`);
}

export function exerciseFromKey(key, content, rng) {
  const [prefix, value] = key.split(':');
  if (prefix === 'mul') {
    const [a, b] = value.split('x').map(Number);
    return { ...mulPair(a, b, rng), srsKey: key };
  }
  if (prefix === 'vert') {
    const m = value.match(/^(\d+)([+-])(\d+)$/);
    if (!m) return null;
    const a = Number(m[1]);
    const b = Number(m[3]);
    const op = m[2];
    return { kind: 'vertical', a, b, op, answer: op === '+' ? a + b : a - b, srsKey: key };
  }
  if (prefix === 'phonics') {
    const pool = phonicsItems(content);
    const item = pool.find((p) => p.letter === value);
    return item ? listenExercise(item, pool, rng) : null;
  }
  if (prefix === 'cvc' || prefix === 'vocab') {
    const groups = prefix === 'cvc' ? Object.values(content.cvc) : Object.values(content.vocab);
    for (const group of groups) {
      const item = group.find((w) => w.word === value);
      if (item) {
        return prefix === 'cvc' ? wordbuildExercise(item, rng) : matchExercise(item, group, rng, prefix);
      }
    }
    return null;
  }
  return null;
}
```

```js
export function canSpeak(synth) {
  return Boolean(synth && typeof synth.speak === 'function');
}

// makeUtterance is injectable for tests; browser default uses SpeechSynthesisUtterance.
export function speak(synth, text, lang = 'en-US', makeUtterance) {
  if (!canSpeak(synth)) return false;
  const make = makeUtterance
    ?? ((t) => new SpeechSynthesisUtterance(t));
  const utterance = make(text);
  utterance.lang = lang;
  utterance.rate = 0.85;
  if (typeof synth.cancel === 'function') synth.cancel();
  synth.speak(utterance);
  return true;
}

const TONES = {
  correct: [[523, 0.12], [784, 0.18]],
  wrong: [[220, 0.2]],
  star: [[659, 0.1], [784, 0.1], [1047, 0.25]],
};

let audioCtx = null;

export function sfx(name) {
  try {
    const Ctx = window.AudioContext || window.webkitAudioContext;
    if (!Ctx) return;
    audioCtx = audioCtx ?? new Ctx();
    let at = audioCtx.currentTime;
    for (const [freq, dur] of TONES[name] ?? []) {
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.frequency.value = freq;
      osc.type = 'sine';
      gain.gain.setValueAtTime(0.15, at);
      gain.gain.exponentialRampToValueAtTime(0.001, at + dur);
      osc.connect(gain).connect(audioCtx.destination);
      osc.start(at);
      osc.stop(at + dur);
      at += dur;
    }
  } catch { /* sound is optional — never break the game */ }
}
```

```js
import { navigate } from '../app.js';
import { buildLevel, exerciseFromKey } from '../levels.js';
import { EXERCISES_PER_LEVEL } from '../config.js';
import { starsForResult, recordLevelResult, updateStreak } from '../engine/progression.js';
import { recordMistake, recordSuccess, dueMistakes } from '../engine/srs.js';
import { localDateISO } from '../storage.js';
import { speak, canSpeak, sfx } from '../audio.js';
import { enqueue } from '../sync.js';

// ---------- pure logic (unit-tested) ----------

export function applyAnswer(session, isCorrect, srsKey) {
  return {
    index: session.index + 1,
    correct: session.correct + (isCorrect ? 1 : 0),
    wrongKeys: isCorrect || session.wrongKeys.includes(srsKey)
      ? session.wrongKeys
      : [...session.wrongKeys, srsKey],
  };
}

export function finishLevel(state, { regionId, levelIndex, correct, total, wrongKeys, todayISO }) {
  const stars = starsForResult({ correct, total });
  let mistakes = state.mistakes;
  for (const key of wrongKeys) mistakes = recordMistake(mistakes, key, todayISO);
  const withQueue = enqueue(state, {
    type: 'session', date: todayISO, region: regionId, level: levelIndex,
    correct, total, stars, wrongKeys,
  });
  return {
    ...withQueue,
    progress: recordLevelResult(state.progress, regionId, levelIndex, stars),
    wallet: { ...state.wallet, stars: state.wallet.stars + stars },
    streak: updateStreak(state.streak, todayISO),
    mistakes,
    lastReward: { stars, regionId, levelIndex },
  };
}

// ---------- DOM shell ----------

export function renderGame(root, ctx, { region, level }) {
  const rng = Math.random;
  const today = localDateISO(new Date());
  // Replay up to 2 due mistakes inside every level (spec §4.6).
  const replays = dueMistakes(ctx.state.mistakes, today)
    .slice(0, 2)
    .map((key) => exerciseFromKey(key, ctx.content, rng))
    .filter(Boolean);
  const fresh = buildLevel(region, level, ctx.content, rng)
    .slice(0, EXERCISES_PER_LEVEL - replays.length);
  const exercises = [...fresh, ...replays];
  let session = { index: 0, correct: 0, wrongKeys: [] };
  let firstTry = true;

  const showExercise = () => {
    if (session.index >= exercises.length) return finish();
    const ex = exercises[session.index];
    firstTry = true;
    root.innerHTML = `
      <header class="row hud">
        <button class="badge" id="quit">✖</button>
        <div class="progress-bar"><div class="progress-fill" style="width:${(session.index / exercises.length) * 100}%"></div></div>
      </header>
      <section class="card game-card" id="stage"></section>
      <p class="feedback" aria-live="polite"></p>`;
    root.querySelector('#quit').addEventListener('click', () => navigate('#map'));
    RENDERERS[ex.kind](root.querySelector('#stage'), ex, handleAnswer(ex));
  };

  const handleAnswer = (ex) => (isCorrect) => {
    const feedback = root.querySelector('.feedback');
    if (isCorrect) {
      sfx('correct');
      // A first-try success on a tracked mistake counts toward graduation.
      if (firstTry) {
        ctx.update({
          ...ctx.state,
          mistakes: recordSuccess(ctx.state.mistakes, ex.srsKey, localDateISO(new Date())),
        });
      }
      session = applyAnswer(session, firstTry, ex.srsKey);
      setTimeout(showExercise, 500);
    } else {
      sfx('wrong');
      if (firstTry) session = { ...session, wrongKeys: applyAnswer(session, false, ex.srsKey).wrongKeys };
      firstTry = false;
      feedback.textContent = 'כמעט! נסי שוב 💪';
    }
  };

  const finish = () => {
    const next = finishLevel(ctx.state, {
      regionId: region, levelIndex: level,
      correct: session.correct, total: exercises.length,
      wrongKeys: session.wrongKeys, todayISO: localDateISO(new Date()),
    });
    ctx.update(next);
    navigate('#reward');
  };

  showExercise();
}

function choiceButtons(stage, labels, isRight, onAnswer) {

}
```

```js
  e.waitUntil(caches.open(CACHE).then((c) => c.addAll(ASSETS)).then(() => self.skipWaiting()));
});

  e.waitUntil(
    caches.keys()
      .then((keys) => Promise.all(keys.filter((k) => k !== CACHE).map((k) => caches.delete(k))))
      .then(() => self.clients.claim()),

});

self.addEventListener('fetch', (e) => {
  if (e.request.method !== 'GET') return; // sync POSTs pass through
  e.respondWith(
    caches.match(e.request, { ignoreSearch: true })
      .then((hit) => hit ?? fetch(e.request)),
  );
});
```

`js/sync.js` (full replacement):
```js
import { SYNC_URL, SYNC_TOKEN } from './config.js';

export const enqueue = (state, event) => ({ ...state, syncQueue: [...state.syncQueue, event] });

export async function flush(queue, postFn) {
  if (queue.length === 0) return { sent: 0, remaining: [] };
  try {
    const ok = await postFn(queue);
    return ok ? { sent: queue.length, remaining: [] } : { sent: 0, remaining: queue };
  } catch {
    return { sent: 0, remaining: queue };
  }
}

// Browser-side poster. text/plain avoids a CORS preflight, which GAS web apps
// do not answer; GAS still receives the body via e.postData.contents.
export async function postToSheet(events) {
  if (!SYNC_URL) return false;
  const res = await fetch(SYNC_URL, {
    method: 'POST',
    headers: { 'Content-Type': 'text/plain;charset=utf-8' },
    body: JSON.stringify({ token: SYNC_TOKEN, events }),
  });
  const body = await res.json().catch(() => ({}));
  return res.ok && body.ok === true;
}
```

```js
// GAYA sync backend. Deploy: New deployment → Web app → Execute as: Me →
// Access: Anyone. Set Script Properties: SYNC_TOKEN (must equal js/config.js
// SYNC_TOKEN), SHEET_ID (target spreadsheet id).

const HEADERS = ['Timestamp', 'Date', 'Region', 'Level', 'Correct', 'Total', 'Stars', 'WrongKeys'];

function doPost(e) {
  try {
    const props = PropertiesService.getScriptProperties();
    const payload = JSON.parse(e.postData.contents);
    if (payload.token !== props.getProperty('SYNC_TOKEN')) {
      return json_({ ok: false, error: 'bad token' });
    }
    const sheet = sessionsSheet_(props.getProperty('SHEET_ID'));
    const rows = (payload.events || []).map(function (ev) {
      return [new Date(), ev.date, ev.region, ev.level, ev.correct, ev.total, ev.stars, (ev.wrongKeys || []).join(', ')];
    });
    if (rows.length > 0) {
      sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, HEADERS.length).setValues(rows);
    }
    return json_({ ok: true, saved: rows.length });
  } catch (err) {
    logError_(err);
    return json_({ ok: false, error: String(err) });
  }
}

function sessionsSheet_(sheetId) {
  const ss = SpreadsheetApp.openById(sheetId);
  let sheet = ss.getSheetByName('Sessions');
  if (!sheet) {
    sheet = ss.insertSheet('Sessions');
    sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]).setFontWeight('bold');
  }
  return sheet;
}

function logError_(err) {
  try {
    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty('SHEET_ID'));
    const sheet = ss.getSheetByName('Errors') || ss.insertSheet('Errors');
    sheet.appendRow([new Date(), String(err)]);
  } catch (ignored) {} // never mask the original failure
}

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

// Run once manually after setting Script Properties: builds the parent summary tab.
function setupSummary() {
  const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty('SHEET_ID'));
  sessionsSheet_(ss.getId());

    ['אזור', 'הצלחה'],
```
